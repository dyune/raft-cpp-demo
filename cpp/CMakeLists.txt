cmake_minimum_required(VERSION 3.20)
project(raft_cpp_demo LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(raft
        src/kv_store.cpp src/transport.cpp src/node.cpp src/cluster.cpp)
target_include_directories(raft PUBLIC src/include)
target_compile_features(raft PUBLIC cxx_std_20)
target_link_libraries(raft PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(raft PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
set(RAFT_SANITIZER "" CACHE STRING "Sanitizer: address or thread (empty disables)")
set_property(CACHE RAFT_SANITIZER PROPERTY STRINGS "" address thread)
if(RAFT_SANITIZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizers require a Clang or GCC toolchain")
    endif()
    if(RAFT_SANITIZER STREQUAL "address")
        set(sanitizers address,undefined)
    elseif(RAFT_SANITIZER STREQUAL "thread")
        set(sanitizers thread)
    else()
        message(FATAL_ERROR "RAFT_SANITIZER must be address, thread, or empty")
    endif()
    target_compile_options(raft PUBLIC -fsanitize=${sanitizers} -fno-omit-frame-pointer)
    target_link_options(raft PUBLIC -fsanitize=${sanitizers})
endif()
add_executable(raft_cpp_demo src/main.cpp)
target_link_libraries(raft_cpp_demo PRIVATE raft)
include(CTest)
if(BUILD_TESTING)
    add_executable(raft_tests tests/tests.cpp)
    target_link_libraries(raft_tests PRIVATE raft)
    # The teaching suite uses assert(), including expressions with side effects.
    if(MSVC)
        target_compile_options(raft_tests PRIVATE /UNDEBUG)
    else()
        target_compile_options(raft_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME raft_tests COMMAND raft_tests)
    set_tests_properties(raft_tests PROPERTIES TIMEOUT 90)
endif()
